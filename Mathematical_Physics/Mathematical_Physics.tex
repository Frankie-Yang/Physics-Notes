\section{Complex Analysis}
\subsection{Analytic Functions}

An analytic function is a complex function that is differentiable at every point in its domain. If a function is differentiable within a region, then we say its analytic throughout the region. There are three nice properties of analytic functions:
\begin{enumerate}
    \item Analytic functions are infinitely differentiable.
    \item The contour integral of an analytic function over a closed curve is zero (Cauchy's theorem).
    \item Analytic continuiation: If two analytic functions agree on a set of point that has an accumulation point, then they are equal on the entire domain.
\end{enumerate}

Another condition for a function to be analytic is that it must satisfy the Cauchy-Riemann equations. If $f(z) = u(x,y) + iv(x,y)$, where $z = x + iy$, then the Cauchy-Riemann equations are given by:
\begin{equation}
\frac{\partial u}{\partial x} = \frac{\partial v}{\partial y}, \quad \frac{\partial u}{\partial y} = -\frac{\partial v}{\partial x}.
\end{equation}

\subsection{Cauchy's Integral Theorem}
\begin{theorem}[Cauchy's Integral Theorem]
 If $f$ is analytic in a simply connected domain $D$, then for any closed contour $C$ in $D$, we have:
\begin{equation}
\oint_C f(z) \, dz = 0.
\end{equation}
\end{theorem}

\begin{proof}
    We first expand $f(z)$ in terms of its real and imaginary parts, then the contour integral can be expressed as:
    \begin{equation}
    \oint_C f(z) \, dz = \oint_C (u + iv)(dx + i dy) = \oint_C (u dx - v dy) + i \oint_C (v dx + u dy).
    \end{equation}
    By Green's theorem, we can convert the line integrals into double integrals over the region $R$ enclosed by $C$:
    \begin{equation}
    \oint_C (u dx - v dy) = \iint_R \left(\frac{\partial (-v)}{\partial x} - \frac{\partial u}{\partial y}\right) dA, \quad \oint_C (v dx + u dy) = \iint_R \left(\frac{\partial u}{\partial x} - \frac{\partial v}{\partial y}\right) dA.
    \end{equation}
    Since $f$ is analytic, it satisfies the Cauchy-Riemann equations,
    \begin{equation}
    \frac{\partial u}{\partial x} = \frac{\partial v}{\partial y}, \quad \frac{\partial u}{\partial y} = -\frac{\partial v}{\partial x}.
    \end{equation}
    Substituting these into the double integrals, we find that both integrals evaluate to zero:
    \begin{equation}
    \oint_C (u dx - v dy) = 0, \quad \oint_C (v dx + u dy) = 0.
    \end{equation}
    Therefore, we conclude that:
    \begin{equation}
    \oint_C f(z) \, dz = 0.
    \end{equation}
\end{proof}

\subsection{Cauchy's Integral Theorem}
\begin{theorem}[Cauchy's Integral Formula]
    If $f$ is analytic in a simply connected domain $D$ and $C$ is a positively orirented simple closed contour in $D$, then for any point $z_0$ inside $C$, we have:
    \begin{equation}
        f(z_0)=\frac{1}{2\pi i} \oint_C \frac{f(z)}{z-z_0} dz.
    \end{equation}
\end{theorem}

\begin{proof}
For an arbitraryly small $\epsilon >0$, we can find a $\delta$ such that for $|z-z_0|<\delta$
\begin{equation}
    f(z)=f(z_0)+f'(z_0)(z-z_0)+\lambda(z)(z-z_0),
\end{equation}

where $|\lambda(z)|<\epsilon$ (the definition of $f(z)$ being analytic at $z_0$). Deforming the contour to be the circle of radius $\delta$ about $z_0$ we find
\begin{equation}
    \begin{split}
        \oint_C\frac{f(z)}{z-z_0}dz &=f(z_0)\oint_C \frac{1}{z-z_0}dz + f'(z_0)\oint_C  dz + \oint_C \lambda(z) dz\\
        &=i2\pi f(z_0) + \oint_C\lambda(z)dz\\
    \end{split}
\end{equation}

As
\begin{equation}
    \left|\oint_C\lambda(z)dz\right|\leq \oint_C |\lambda(z)||dz| < \epsilon 2\pi \delta,
\end{equation}
we have
\begin{equation}
    \left|\oint_C\frac{f(z)}{z-z_0}dz - 2\pi i f(z_0)\right| < \epsilon 2\pi \delta.
\end{equation}

Since $\epsilon$ is arbitrarily small, we have
\begin{equation}
    \oint_C\frac{f(z)}{z-z_0}dz = 2\pi i f(z_0).
\end{equation}
\end{proof}

\subsection{Derivatives from Cauchy's Integral Theorem}

Takes derivative of the Cauchy's Integral Formula with respect to $z_0$, we have
\begin{equation}
    f'(z_0)=\frac{1}{2\pi i}\frac{d}{d z_0}\oint_C\frac{f(z)}{z-z_0}dz = \frac{1}{2\pi i}\oint_C\frac{f(z)}{(z-z_0)^2}dz.
\end{equation}

Then more genrally we will have the nth order derivative of $f$ at $z_0$ as
\begin{equation}
    f^{(n)}(z_0)=\frac{n!}{2\pi i}\oint_C\frac{f(z)}{(z-z_0)^{n+1}}dz.
\end{equation}

\subsection{Liouville's Theorem}
\begin{theorem}[Liouville's Theorem]
    If $f$ is entire and bounded, then $f$ is constant.
\end{theorem}

\begin{proof}
    If $f$ is entire and bounded, then there exist a constant $M$ such that $|f(z)| \leq M$ in the entire complex plane. By Cauchy's integral formula for the first derivative, we have:
    \begin{equation}
        f'(z) = \frac{1}{2\pi i} \oint_C \frac{f(w)}{(w-z)^2} dw,
    \end{equation}
    
    where $C$ is a circle of radius $R$ centered at $z$. Taking the modulus and using the bound on $f$, we get:
    \begin{equation}
        |f'(z)| \leq \frac{1}{2\pi} \oint_C \frac{|f(w)|}{|w-z|^2} |dw| \leq \frac{1}{2\pi} \cdot M \cdot \frac{2\pi R}{R^2} = \frac{M}{R}.
    \end{equation}
    
    As $R \to \infty$, we have $|f'(z)| \to 0$, which means $f'(z) = 0$ for all $z$. Therefore, $f$ is constant.
\end{proof}

\subsection{Laurent Expansion}

This is a very important theorem in complex analysis to express an analytic funcition in the form of power series. The Laurent expansion is a representation of a complex function as a series that includes both positive and negative powers of the variable. It is particularly useful for functions that have singularities, as it allows us to express the function in a neighborhood of those singularities. 
\begin{theorem}[Laurent Expansion]
    If $f$ is analytic in an annulus $r < |z - z_0| < R$, then $f$ has a unique representation as a Laurent series:
    \begin{equation}
        f(z) = \sum_{n=-\infty}^{\infty} a_n (z - z_0)^n,
    \end{equation}
    where the coefficients are given by:
    \begin{equation}
        a_n = \frac{1}{2\pi i} \oint_C \frac{f(w)}{(w - z_0)^{n+1}} dw,
    \end{equation}
    and $C$ is any simple closed contour in the annulus, oriented positively.
\end{theorem}

If the function $f$ is analytic throughout the whole disk, then the Laurent series reduces to a Taylor series. If $f$ has a singularity at $z_0$, then the Laurent series will have negative powers of $(z-z_0)$. The principal part of the Laurent series is the sum of the terms with negative powers, and it captures the behavior of the function near the singularity.
\begin{proof}
    If $f$ is analytic in the annulus $ r < |z-z_0|<R$, then the Cauchy integral formula gives us:
    \begin{equation}
        f(z) = \frac{1}{2\pi i}\oint_{C_1} \frac{f(w)}{w-z} dw - \frac{1}{2\pi i}\oint_{C_2} \frac{f(w)}{w-z} dw,
    \end{equation}

    where $C_1$ is a positively oriented contour around $z_0$ with radius $R$, and $C_2$ is a negatively oriented contour around $z_0$ with radius $r$. We can expand the integrands in geometric series:
    \begin{equation}
        \frac{1}{w-z} = \frac{1}{w-z_0 - (z-z_0)} = \sum_{n=0}^{\infty} \frac{(z-z_0)^n}{(w-z_0)^{n+1}} \quad \text{for } |z-z_0| < |w-z_0|,
    \end{equation}

    For the second term, we have:
    \begin{equation}
        -\frac{1}{w-z} = -\frac{1}{(z-z_0) - (w-z_0)} = \sum_{m=0}^{\infty} \frac{(w-z_0)^{m}}{(z-z_0)^{m+1}} \quad \text{for } |z-z_0| > |w-z_0|.
    \end{equation}
    Then we substitue $n = -m-1$ in the second series, we have
    \begin{equation}
        -\frac{1}{w-z} = \sum_{n=-\infty}^{-1} \frac{(w-z_0)^{-n-1}}{(z-z_0)^{-n}}.
    \end{equation}

    Putting them together, we have
    \begin{equation}
        f(z) = \sum_{n=0}^{\infty} \left( \frac{1}{2\pi i} \oint_{C_1} \frac{f(w)}{(w-z_0)^{n+1}} dw \right) (z-z_0)^n + \sum_{n=-\infty}^{-1} \left( \frac{1}{2\pi i} \oint_{C_2} \frac{f(w)}{(w-z_0)^{n+1}} dw \right) (z-z_0)^n.
    \end{equation}

    Then this is the Laurent expansion of $f(z)$ in the annulus $r < |z-z_0| < R$,
    \begin{equation}
        f(z) = \sum_{n=-\infty}^{\infty} a_n (z - z_0)^n,
    \end{equation}
\end{proof}

\subsection{Isolated Singularities}
\begin{definition}[Isolated Singularity]
    A point $z_0 \in \mathbb{C}$ is called an isolated singularity of a complex function $f(z)$ if there exist $r > 0$ such that $f(z)$ is analytic in the punctured disk $0 < |z - z_0| < r$, but $f(z)$ is not analytic at $z_0$.
\end{definition}

\begin{definition}[Removable Singularity]
    A removable singularity is an isolated singularity of a complex function $f(z)$ at a point $z_0$ such that the limit $\lim_{z \to z_0} f(z)$ exists and is finite. In this case, we can define a new function $\tilde{f}(z)$ that is equal to $f(z)$ for $z \neq z_0$ and is defined to be $\tilde{f}(z_0) = \lim_{z \to z_0} f(z)$. The new function $\tilde{f}(z)$ is then analytic at $z_0$, effectively "removing" the singularity.
\end{definition}
\begin{definition}[Pole]
  A pole of order m is an isolated singularity of a complex function $f(z)$ at a point $z_0$ such that the Laurent series expansion of $f(z)$ around $z_0$ has $a_{-m} \neq 0$ and $a_{-n} = 0$ for $n > m$.
\end{definition}

In particular, the Laurent expansion for a order of m pole of complex function $f(z)$ in the region $0 < |z-z_0| < R$ is given by
\begin{equation}
    f(z)=\sum_{k=0}^{\infty} a_{k} (z-z_0)^k + \frac{a_{-1}}{z-z_0} + \frac{a_{-2}}{(z-z_0)^2} + \cdots + \frac{a_{-m}}{(z-z_0)^m}.
\end{equation}
\begin{definition}{Essential Singularity}
  An essential singularity is a type of isolated singularity of a complex function where the Laurent expansion of the function around that point contains infinitely many negative power terms.
\end{definition}

Then the Laurent expansion for a complex function $f(z)$ about its essential singularity is
\begin{equation}
    f(z)=\sum_{k=0}^{\infty} a_{k} (z-z_0)^k + \sum_{n=-\infty}^{-1} a_n (z-z_0)^n, \quad  a_{n} \neq 0 ~ \text{for infinitely many }n
\end{equation}
